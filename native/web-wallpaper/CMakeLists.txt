cmake_minimum_required(VERSION 3.20)
project(better-wallpaper-web LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.6 REQUIRED COMPONENTS Quick WebEngineQuick)
find_package(LayerShellQt REQUIRED)
qt_add_executable(better-wallpaper-web main.cpp)
qt_add_resources(better-wallpaper-web "wallpaper" PREFIX "/" FILES WebWallpaper.qml)
set(WEB_UI "${CMAKE_CURRENT_SOURCE_DIR}/../../kde/org.better-wallpaper/contents/ui")
qt_add_resources(better-wallpaper-web "web-protocol" PREFIX "/" BASE "${WEB_UI}" FILES "${WEB_UI}/WebSurface.qml" "${WEB_UI}/WebBridge.js")
target_link_libraries(better-wallpaper-web PRIVATE Qt6::Quick Qt6::WebEngineQuick LayerShellQt::Interface)
target_compile_options(better-wallpaper-web PRIVATE -Wall -Wextra -Werror)
